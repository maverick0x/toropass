# Step 1: Create the application

curl -X POST http://localhost:3000/api/v1/oauth/apps/register \
  -H "Content-Type: application/json" \
  -H "x-api-key: your_super_secret_flutter_api_key_here" \
  -d '{
    "name": "DeFi Swap Platform",
    "developerId": "xxxxxx-xxxxxxx-xxxxxxx-xxxxxxxx",
    "redirectUri": "https://example.com"
  }'

RESPONSE: {
  "status":"success",
  "message":"Developer credentials generated successfully. Save the client secret safely!",
  "data":{
    "id":"xxxxxx-xxxxxxx-xxxxxxx-xxxxxxxx",
    "name":"DeFi Swap Platform",
    "clientId":"xxxxxx-xxxxxxx-xxxxxxx-xxxxxxxx",
    "clientSecret":"xxxxxx-xxxxxxx-xxxxxxx-xxxxxxxx","
    redirectUri":"https://example.com"
  }
}


# Step 2: Simulate the User Consent (Inside ToroPass App)

curl -X POST http://localhost:3000/api/v1/oauth/authorize \
  -H "Content-Type: application/json" \
  -H "x-api-key: your_super_secret_flutter_api_key_here" \
  -d '{
    "client_id": "xxxxxx-xxxxxxx-xxxxxxx-xxxxxxxx",
    "user_id": "xxxxxx-xxxxxxx-xxxxxxx-xxxxxxxx",
    "redirect_uri": "https://example.com",
    "scopes": ["kyc_status", "wallet"]
  }'

RESPONSE: {
  "status":"success",
  "data":{
    "code":"1111111111111111111111"
  }
}
